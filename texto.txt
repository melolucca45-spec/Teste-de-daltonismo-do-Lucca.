<script>
    // Configuração das imagens e diagnósticos de erro
    const dados = [
        { url: "https://i.ibb.co/3ykS8Vz/daltonismo1.png", correta: "2", tipo: "Daltonismo Vermelho/Verde" },
        { url: "https://i.ibb.co/60qYvjF/daltonismo2.png", correta: "29", tipo: "Deuteranopia (Verde)" },
        { url: "https://i.ibb.co/jH8S5mF/daltonismo3.png", correta: "42", tipo: "Protanopia (Vermelho)" },
        { url: "https://i.ibb.co/q9WJ4T2/daltonismo4.png", correta: "73", tipo: "Daltonismo Geral" },
        { url: "https://i.ibb.co/8m0N33k/daltonismo5.png", correta: "74", tipo: "Daltonismo Vermelho/Verde" },
        { url: "https://i.ibb.co/7XkP7Vz/daltonismo6.png", correta: "12", tipo: "Teste de Controle (Todos devem ver)" },
        { url: "https://i.ibb.co/LpYx0pP/daltonismo7.png", correta: "5", tipo: "Deuteranopia / Protanopia" },
        { url: "https://i.ibb.co/zH9v4Tf/daltonismo8.png", correta: "96", tipo: "Daltonismo Geral" },
        { url: "https://i.ibb.co/v3m6H4z/daltonismo9.png", correta: "45", tipo: "Daltonismo Vermelho/Verde" }
    ];

    let atual = 0;
    let acertos = 0;

    const imgTag = document.getElementById('imagem-atual');
    const inputTag = document.getElementById('resposta');
    const feedbackTag = document.getElementById('feedback-caixa');
    const contadorTag = document.getElementById('contador');

    function carregar() {
        imgTag.src = dados[atual].url;
        contadorTag.innerText = `Imagem ${atual + 1} de ${dados.length}`;
        inputTag.value = "";
        feedbackTag.style.display = "none";
        document.getElementById('btn-proximo').innerText = "Verificar";
    }

    function validar() {
        const btn = document.getElementById('btn-proximo');
        
        if (btn.innerText === "Próxima Imagem") {
            atual++;
            if (atual < dados.length) {
                carregar();
            } else {
                finalizar();
            }
            return;
        }

        const userResp = inputTag.value.trim();
        if (!userResp) return;

        feedbackTag.style.display = "block";

        if (userResp === dados[atual].correta) {
            acertos++;
            feedbackTag.className = "feedback acerto";
            feedbackTag.innerText = "Correto!";
        } else {
            feedbackTag.className = "feedback erro";
            feedbackTag.innerText = `Incorreto. Isso pode indicar: ${dados[atual].tipo}`;
        }

        btn.innerText = "Próxima Imagem";
    }

    function finalizar() {
        document.getElementById('tela-teste').classList.add('hidden');
        document.getElementById('tela-final').classList.remove('hidden');
        document.getElementById('placar').innerText = `${acertos} / ${dados.length}`;
        
        let texto = acertos === dados.length ? "Sua visão de cores é excelente!" : "Você teve dificuldade em algumas cores. Consulte um médico para um exame completo.";
        document.getElementById('diagnostico-final').innerText = texto;
    }

    carregar();
</script>
